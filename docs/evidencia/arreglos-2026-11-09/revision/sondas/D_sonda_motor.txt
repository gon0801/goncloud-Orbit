activa_desde_settings(''): ValueError: setting ads_proteccion_venta: debe ser 'proteccion_venta_v1', llego ''
activa_desde_settings(True): ValueError: setting ads_proteccion_venta: debe ser 'proteccion_venta_v1', llego True
activa_desde_settings('proteccion_venta_v1'): OK -> True
economia_desde_settings(None): OK -> 'auto03'

economia_desde_settings('producto_directo_v1'): OK -> 'producto_directo_v1'
economia_cambio_desde_settings(None): OK -> False
