import os,json,datetime,psycopg
from psycopg.rows import dict_row
from app.db import connect
queries={'episodios': "SELECT id,profile_id,platform,tipo,opened_at,opened_run_id,alert_sent_at,recovered_at,recovered_run_id,recovery_sent_at,closed_at FROM ads_ingest_incident WHERE opened_at >= '2026-09-30' ORDER BY id", 'recovery_sin_soporte': "SELECT i.id FROM ads_ingest_incident i LEFT JOIN ingest_run r ON r.id=i.recovered_run_id WHERE i.recovered_at IS NOT NULL AND (r.id IS NULL OR r.ok IS DISTINCT FROM true OR r.source <> 'amazon_ads_reports_v3' OR NOT EXISTS (SELECT 1 FROM ads_report_result a WHERE a.ingest_run_id=r.id AND a.status='written' AND (i.profile_id IS NULL OR a.profile_id=i.profile_id) AND (i.platform IS NULL OR a.platform=i.platform)))", 'principal_7d': "SELECT (r.finished_at AT TIME ZONE 'UTC')::date AS dia,a.profile_id,a.platform,count(DISTINCT r.id) AS corridas_ok,count(*) AS reportes_written,min(r.finished_at) AS primer_fin_ok,max(r.finished_at) AS ultimo_fin_ok FROM ingest_run r JOIN ads_report_result a ON a.ingest_run_id=r.id WHERE r.source='amazon_ads_reports_v3' AND r.ok=true AND a.status='written' AND r.finished_at >= '2026-10-29 00:00+00' AND r.finished_at < '2026-11-05 00:00+00' GROUP BY 1,2,3 ORDER BY 1,2,3", 'pausa': 'SELECT d.id,d.ad_entity_id,d.decided_at,d.window_start,d.window_end,d.inputs,d.config_version_id,e.platform,e.kind,e.external_id,a.attempted_at,a.confirmed_at,a.verify_ok,a.platform_ack FROM decision d JOIN ad_entity e ON e.id=d.ad_entity_id JOIN decision_application a ON a.decision_id=d.id WHERE d.id=2736', 'bids_previos': 'SELECT d.id,d.kind,d.decided_at,d.old_value,d.new_value,a.confirmed_at,a.verify_ok FROM decision d LEFT JOIN decision_application a ON a.decision_id=d.id WHERE d.ad_entity_id=(SELECT ad_entity_id FROM decision WHERE id=2736) AND d.decided_at <= (SELECT decided_at FROM decision WHERE id=2736) ORDER BY d.decided_at DESC LIMIT 15'}
out={"consultado_utc":datetime.datetime.now(datetime.UTC).isoformat(),"modo":"ORBIT_DSN_READ; BEGIN READ ONLY; statement_timeout=15s","consultas":{}}
with connect(os.environ["ORBIT_DSN_READ"],row_factory=dict_row) as c:
 c.execute("BEGIN READ ONLY")
 c.execute("SET LOCAL statement_timeout=15000")
 for nombre,sql in queries.items():
  out["consultas"][nombre]={"sql":sql,"filas":c.execute(sql).fetchall()}
print(json.dumps(out,default=str,ensure_ascii=False,indent=2))
