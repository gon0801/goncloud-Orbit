SELECT d.decision_date, count(*) AS decisiones,
       max(e.observed_at) AS ultima_estimacion, max(o.fetched_at) AS oferta_mas_nueva,
       to_char(min(d.created_at) - max(o.fetched_at), 'HH24:MI:SS') AS edad_oferta_al_decidir
  FROM precio_decision d
  JOIN LATERAL (SELECT e.* FROM estimacion_escenario e
                 WHERE e.listing_id = d.listing_id AND e.observed_at <= d.created_at
                 ORDER BY e.observed_at DESC, e.id DESC LIMIT 1) e ON true
  LEFT JOIN estimacion_oferta_observation o ON o.id = e.oferta_observation_id
 WHERE d.platform = 'amazon_us' AND d.decision_date >= '2026-10-30'
   AND d.created_at::time < '14:00'
 GROUP BY 1 ORDER BY 1
